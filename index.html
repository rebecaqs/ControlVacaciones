""" vacaciones_manana.py Lee el Excel de vacaciones (alojado en SharePoint y sincronizado por OneDrive a una carpeta local) y muestra quién tiene vacaciones el día siguiente a la fecha en la que se ejecuta el script. Estructura esperada del Excel (una fila de cabecera de meses + dos filas de calendario, y una fila por persona): Fila 2 -> nombre del mes, en celdas combinadas (p.ej. "SEPTIEMBRE") Fila 3 -> letra del día de la semana (L, M, X, J, V, S, D) Fila 4 -> número de día del mes (1, 2, 3, ...) Fila 5+ -> una fila por persona. Columnas fijas: A = Departamento B = Ubicación C = Persona D = Horas Trabajadas E = Días NO facturados F..I = resumen de vacaciones (Actual/Ant./Cons/Pdtes) J en adelante = calendario, un día por columna. En el calendario, el código "V" significa "Vacaciones". Si la estructura de tu Excel cambia de un año a otro, ajusta las constantes en la sección CONFIGURACIÓN. """ from __future__ import annotations import argparse import os import sys import unicodedata from datetime import date, timedelta import openpyxl from openpyxl.utils import get_column_letter # --------------------------------------------------------------------------- # CONFIGURACIÓN — ajusta esto si tu Excel tiene una estructura distinta # --------------------------------------------------------------------------- # Ruta al Excel. Por defecto se lee de la variable de entorno # VACACIONES_XLSX_PATH; si no existe, usa la ruta de ejemplo de abajo. # En Windows, la carpeta sincronizada de SharePoint/OneDrive suele tener # una ruta como: # C:\Users\tu.usuario\OneDrive - TuEmpresa\Vacaciones\Vacaciones2026.xlsx DEFAULT_XLSX_PATH = r"C:\Users\tu.usuario\OneDrive - TuEmpresa\Vacaciones\Vacaciones2026.xlsx" FILA_MESES = 2 # fila con el nombre del mes (celdas combinadas) FILA_DIA_SEMANA = 3 # fila con la letra del día de la semana FILA_NUM_DIA = 4 # fila con el número de día del mes PRIMERA_FILA_PERSONAS = 5 # primera fila donde puede empezar una persona COL_DEPARTAMENTO = 1 # columna A COL_UBICACION = 2 # columna B COL_PERSONA = 3 # columna C COL_HORAS_TRABAJADAS = 4 # columna D -> se usa para saber si la fila es una persona real CODIGO_VACACIONES = "V" MESES_ES = [ "ENERO", "FEBRERO", "MARZO", "ABRIL", "MAYO", "JUNIO", "JULIO", "AGOSTO", "SEPTIEMBRE", "OCTUBRE", "NOVIEMBRE", "DICIEMBRE", ] # --------------------------------------------------------------------------- def normalizar(texto: str) -> str: """Quita acentos y pone en mayúsculas, para comparar nombres de mes.""" if texto is None: return "" texto = str(texto).strip().upper() texto = "".join( c for c in unicodedata.normalize("NFD", texto) if unicodedata.category(c) != "Mn" ) return texto def obtener_ruta_excel() -> str: return os.environ.get("VACACIONES_XLSX_PATH", DEFAULT_XLSX_PATH) def tiene_bloques_de_mes(ws) -> bool: """True si la hoja tiene en la fila de meses celdas combinadas con nombres de mes (es decir, parece la hoja de calendario de vacaciones).""" return any(b[2] in MESES_ES for b in bloques_de_mes(ws)) def seleccionar_hoja(wb, nombre: str | None = None): """Elige la hoja del calendario de vacaciones. - Si se indica `nombre`, se usa esa hoja. - Si no, se prueban las hojas cuyo nombre contiene 'vacaciones' y, si ninguna sirve, cualquier hoja que tenga los bloques de meses. """ if nombre: if nombre not in wb.sheetnames: raise ValueError( f"No existe la hoja '{nombre}'. Hojas disponibles: {', '.join(wb.sheetnames)}" ) return wb[nombre] por_nombre = [n for n in wb.sheetnames if "VACACIONES" in normalizar(n)] for n in por_nombre + [n for n in wb.sheetnames if n not in por_nombre]: if tiene_bloques_de_mes(wb[n]): return wb[n] raise ValueError( "No se encontró ninguna hoja con el calendario de vacaciones " f"(hojas del libro: {', '.join(wb.sheetnames)}). Indica la hoja con --hoja." ) def detectar_anio_hoja(ws, ruta_excel: str) -> int: """Intenta averiguar a qué año corresponde la hoja (para el caso especial de diciembre -> enero del año siguiente).""" for fuente in (ws.title, os.path.basename(ruta_excel)): digitos = "".join(c for c in fuente if c.isdigit()) if len(digitos) >= 4: return int(digitos[:4]) return date.today().year def bloques_de_mes(ws) -> list[tuple[int, int, str]]: """Devuelve [(columna_inicio, columna_fin, 'NOMBRE_MES'), ...] en el orden en que aparecen en la hoja (de izquierda a derecha).""" bloques = [] for mc in ws.merged_cells.ranges: if mc.min_row == FILA_MESES and mc.max_row == FILA_MESES: valor = ws.cell(row=FILA_MESES, column=mc.min_col).value bloques.append((mc.min_col, mc.max_col, normalizar(valor))) bloques.sort(key=lambda b: b[0]) return bloques def localizar_columna_dia(ws, fecha_objetivo: date, anio_hoja: int) -> int: """Busca la columna del calendario que corresponde a `fecha_objetivo`.""" nombre_mes = MESES_ES[fecha_objetivo.month - 1] bloques = bloques_de_mes(ws) candidatos = [b for b in bloques if b[2] == nombre_mes] if not candidatos: raise ValueError(f"No se encontró el mes '{nombre_mes}' en la hoja '{ws.title}'.") if fecha_objetivo.year == anio_hoja: bloque = candidatos[0] elif fecha_objetivo.year == anio_hoja + 1: # Caso especial: 31 de diciembre -> 1 de enero del año siguiente. # Si la hoja incluye un bloque extra de enero al final, es ese. bloque = candidatos[-1] else: raise ValueError( f"La fecha {fecha_objetivo} no corresponde al año de la hoja ({anio_hoja})." ) col_inicio, col_fin, _ = bloque for col in range(col_inicio, col_fin + 1): if ws.cell(row=FILA_NUM_DIA, column=col).value == fecha_objetivo.day: return col raise ValueError( f"No se encontró el día {fecha_objetivo.day} dentro del bloque de {nombre_mes}." ) def personas_con_vacaciones(ws, columna_dia: int) -> list[dict]: """Recorre las filas de personas y devuelve las que tienen 'V' en la columna del día indicado.""" resultado = [] departamento_actual = None fila = PRIMERA_FILA_PERSONAS filas_vacias_seguidas = 0 while fila <= ws.max_row: horas = ws.cell(row=fila, column=COL_HORAS_TRABAJADAS).value nombre = ws.cell(row=fila, column=COL_PERSONA).value es_fila_persona = isinstance(horas, (int, float)) if es_fila_persona: filas_vacias_seguidas = 0 depto = ws.cell(row=fila, column=COL_DEPARTAMENTO).value if depto: departamento_actual = depto valor_dia = ws.cell(row=fila, column=columna_dia).value if isinstance(valor_dia, str) and valor_dia.strip().upper() == CODIGO_VACACIONES: resultado.append({ "persona": nombre, "departamento": departamento_actual, }) else: filas_vacias_seguidas += 1 # Tras varias filas vacías seguidas asumimos que ya hemos # pasado la tabla de personas (lo que sigue es la leyenda). if filas_vacias_seguidas >= 5: break fila += 1 return resultado def main() -> int: parser = argparse.ArgumentParser( description="Muestra quién tiene vacaciones el día siguiente, según el Excel de vacaciones." ) parser.add_argument( "--fecha", help="Fecha de referencia en formato YYYY-MM-DD (por defecto: hoy). " "Se comprobará el día siguiente a esta fecha. Útil para pruebas.", ) parser.add_argument( "--excel", help="Ruta al archivo .xlsx. Si no se indica, se usa la variable de " "entorno VACACIONES_XLSX_PATH o la ruta configurada en el script.", ) parser.add_argument( "--hoja", help="Nombre de la hoja con el calendario de vacaciones. Si no se indica, " "se detecta automáticamente.", ) args = parser.parse_args() hoy = date.fromisoformat(args.fecha) if args.fecha else date.today() manana = hoy + timedelta(days=1) ruta_excel = args.excel or obtener_ruta_excel() if not os.path.exists(ruta_excel): print(f"ERROR: no se encuentra el archivo Excel en:\n {ruta_excel}") print("Comprueba que la ruta es correcta o define la variable de entorno " "VACACIONES_XLSX_PATH.") return 1 try: wb = openpyxl.load_workbook(ruta_excel, data_only=True) except PermissionError: print("ERROR: no se puede abrir el Excel (está bloqueado). Ciérralo en " "Excel o espera a que OneDrive termine de sincronizar, y vuelve a probar.") return 1 try: ws = seleccionar_hoja(wb, args.hoja) except ValueError as e: print(f"ERROR: {e}") return 1 anio_hoja = detectar_anio_hoja(ws, ruta_excel) try: columna_dia = localizar_columna_dia(ws, manana, anio_hoja) except ValueError as e: print(f"ERROR: {e}") return 1 personas = personas_con_vacaciones(ws, columna_dia) print(f"Vacaciones para el {manana.strftime('%d/%m/%Y')} " f"({get_column_letter(columna_dia)}, hoja '{ws.title}')") print("-" * 60) if not personas: print("Nadie tiene vacaciones ese día.") else: for p in personas: depto = f" [{p['departamento']}]" if p["departamento"] else "" print(f" - {p['persona']}{depto}") return 0 if __name__ == "__main__": sys.exit(main())
